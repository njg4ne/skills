% Appendix heading reading "Appendix A: Title" in the heading, contents, and bookmarks,
% with \autoref, \ref, and \nameref still working. Use after \appendix.
\newcommand{\appsection}[2]{%
  \stepcounter{section}%
  \section*{\appendixname~\thesection: #1}%
  \addtocounter{section}{-1}\refstepcounter{section}%
  \def\@currentlabelname{#1}%
  \addcontentsline{toc}{section}{\appendixname~\thesection: #1}%
  \label{#2}}
